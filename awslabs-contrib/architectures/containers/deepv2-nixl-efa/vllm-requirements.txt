# bytes, so it lists both). pip refuses any file whose hash is not listed.

# manylinux_2_35 (Ubuntu 22.04) together with torch==2.14.0+cu130.

# and pip must never replace them:
#   torch==2.14.0+cu130       the Dockerfile's torch layer (TORCH_VERSION)

#                             (torch itself pins an older NCCL without the GIN API)

#   printf 'vllm @ <VLLM_WHEEL_URL>\ntorch==<TORCH_VERSION>+cu130\n' > req.in
#   printf 'nvidia-nccl-cu13==2.31.2\n' > overrides.txt

#     --index-url https://pypi.org/simple --extra-index-url https://download.pytorch.org/whl/cu130 \
#     --index-strategy unsafe-best-match --override overrides.txt --no-header --no-annotate -o lock.txt
#   grep -vE '^(torch==|nvidia-nccl-cu13==|vllm @)' lock.txt > pins.txt
#   uv run --no-project --with packaging==25.0 python trim_lock_hashes.py pins.txt hashed.txt
